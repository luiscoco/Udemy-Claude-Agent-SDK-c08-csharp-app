Step 0 · Prepare the application before the workflow

Use your own Anthropic API key and an existing agent ID and environment ID.

For the optional live path, set your API key, then run npm run setup if you do not already have resource IDs. The script creates the environment before a text-only agent, saves each ID immediately, and reuses IDs already configured. /** Optional one-time provisioning. Reuses IDs already present in .env. */
import Anthropic from "@anthropic-ai/sdk";
import { existsSync, readFileSync, writeFileSync } from "node:fs";
if (
  !process.env.ANTHROPIC_API_KEY ||
  process.env.ANTHROPIC_API_KEY.startsWith("your_")
) {
  throw new Error("Set your own ANTHROPIC_API_KEY in .env first.");
}
const client = new Anthropic({ maxRetries: 0 });
let contents = existsSync(".env") ? readFileSync(".env", "utf8") : "";
function save(key: string, value: string) {
  const pattern = new RegExp("^\\s*" + key + "=.*$", "m");
  contents = pattern.test(contents)
    ? contents.replace(pattern, key + "=" + value)
    : contents.trimEnd() + "\n" + key + "=" + value + "\n";
  writeFileSync(".env", contents, { mode: 0o600 });
}
const valid = (value: string | undefined) =>
  Boolean(value && !value.startsWith("your_"));
if (!valid(process.env.CLAUDE_MANAGED_ENVIRONMENT_ID)) {
  const environment = await client.beta.environments.create({
    name: "C08 student environment",
    config: { type: "cloud", networking: { type: "limited" } },
  });
  // Save immediately so a later agent failure does not lose the environment ID.
  save("CLAUDE_MANAGED_ENVIRONMENT_ID", environment.id);
}
if (!valid(process.env.CLAUDE_MANAGED_AGENT_ID)) {
  const agent = await client.beta.agents.create({
    name: "C08 text-only student agent",
    model: process.env.CLAUDE_MODEL || "claude-opus-5-5",
    system: "Answer the student's question clearly and briefly.",
    tools: [],
    mcp_servers: [],
    skills: [],
  });
  save("CLAUDE_MANAGED_AGENT_ID", agent.id);
}
console.log(
  "Prerequisite IDs are saved in your local .env. Existing configured IDs were reused.",
);

const $ = (s) => document.querySelector(s),
  button = $("#run"),
  status = $("#status");
fetch("/api/config")

button.addEventListener("click", async () => {
  const prompt = $("#prompt").value.trim();
  if (!prompt) {
    status.textContent = "Enter a prompt.";
    return;
  }
  button.disabled = true;
  status.textContent = "Creating session…";
  $("#empty").hidden = true;
  $("#result").hidden = false;
  $("#session").textContent = "—";
  $("#stop").textContent = "Waiting for evidence";
  $("#events").replaceChildren();
  let stopped = false,
    failed = false;
  const handle = ({ type, value }) => {
    if (type === "session") {
      $("#session").textContent = value;
      status.textContent = "Routing live events…";
    } else if (type === "route") {
      const li = document.createElement("li"),
        tag = document.createElement("span"),
        detail = document.createElement("details"),
        summary = document.createElement("summary"),
        pre = document.createElement("pre");
      tag.className = "tag";
      tag.textContent = value.sink;
      li.append(tag, document.createTextNode(value.event.type));
      summary.textContent = "Inspect event";
      pre.textContent = JSON.stringify(value.event, null, 2);
      detail.append(summary, pre);
      li.append(detail);
      $("#events").append(li);
    } else if (type === "stop") {
      stopped = true;
      $("#stop").textContent = value;
      status.textContent = "Observation finished.";
    } else if (type === "error") {
      failed = true;
      $("#stop").textContent = "Error; outcome unconfirmed";
      status.textContent = value;
    }
  };
  try {
    const r = await fetch("/api/run", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ prompt }),
    });
    if (!r.ok) {
      const d = await r.json();
      throw new Error(d.error || "Request failed");
    }
    const reader = r.body.getReader(),
      decoder = new TextDecoder();
    let pending = "";
    while (true) {
      const { value, done } = await reader.read();
      pending += decoder.decode(value || new Uint8Array(), { stream: !done });
      const lines = pending.split("\n");
      pending = lines.pop();
      for (const line of lines) if (line) handle(JSON.parse(line));
      if (done) break;
    }
    if (pending.trim()) handle(JSON.parse(pending));
    if (!stopped && !failed) {
      $("#stop").textContent = "Incomplete";
      status.textContent = "Stream ended without a terminal event.";
    }
  } catch (e) {
    status.textContent = e.message;
    $("#stop").textContent = "Error; outcome unconfirmed";
  } finally {
    button.disabled = false;
  }
});

2 · Execute the provider workflow · part 1

The Node host validates input, uses configured resources, and returns observable evidence. Read the stream ordering and failure branches.

Source: src/server.ts · starting at line 1

import { serveStudentAsset } from "./student-assets";
import http from "node:http";
import { readFile } from "node:fs/promises";
import { existsSync } from "node:fs";
import path from "node:path";
import Anthropic from "@anthropic-ai/sdk";
import { createRouter, type RoutedEvent } from "./router";
if (existsSync(".env")) process.loadEnvFile(".env");
const port = Number(process.env.PORT || 3008),
  key = process.env.ANTHROPIC_API_KEY || "",
  agent = process.env.CLAUDE_MANAGED_AGENT_ID || "",
  environment = process.env.CLAUDE_MANAGED_ENVIRONMENT_ID || "";
const configured = [key, agent, environment].every((v) => v && !v.startsWith("your_"));
http
  .createServer(async (req, res) => {
    res.setHeader("Cache-Control", "no-store");
    try {
      if (await serveStudentAsset(req, res)) return;
      if (req.method === "GET" && req.url === "/api/config") {
        res.writeHead(200, { "Content-Type": "application/json" });
        res.end(JSON.stringify({ configured }));
        return;
      }
      if (req.method === "POST" && req.url === "/api/run") {
        const origin = req.headers.origin;
        if (
          origin &&
          origin !== `http://127.0.0.1:${port}` &&
          origin !== `http://localhost:${port}`
        ) {
          res.writeHead(403).end();
          return;
        }
        let raw = "";
        for await (const chunk of req) {
          raw += chunk;
          if (raw.length > 10000) throw new Error("Request too large");
        }
        const body = JSON.parse(raw || "{}");
        if (
          typeof body.prompt !== "string" ||
          !body.prompt.trim() ||
          body.prompt.length > 2000
        )
          throw new Error("Enter a prompt of up to 2,000 characters.");
        if (!configured) throw new Error("Configure the API key and IDs in .env.");
        res.writeHead(200, { "Content-Type": "application/x-ndjson; charset=utf-8" });
        const send = (type: string, value: unknown) => {
          if (!res.destroyed) res.write(JSON.stringify({ type, value }) + "\n");
        };
        try {
          const client = new Anthropic({ apiKey: key }),
            session = await client.beta.sessions.create({
              agent,
              environment_id: environment,
              budget: {
                type: "limit",
                max_list_cost: { amount: "50", currency: "USD" },
              },
            });
          send("session", session.id);
          const stream = await client.beta.sessions.events.stream(session.id);
          res.on("close", () => stream.controller.abort());
          await client.beta.sessions.events.send(session.id, {
            events: [
              {
                type: "user.message",
                content: [{ type: "text", text: body.prompt.trim() }],
              },
            ],
          });
          const route = createRouter();
          for await (const event of stream) {
            await route(event as unknown as RoutedEvent, (sink, e) =>
              send("route", { sink, event: e }),
            );
            if (event.type === "session.status_idle") {
              send("stop", event.stop_reason.type);
              break;
            }
            if (event.type === "session.status_terminated") {
              send("stop", "terminated");
              break;
            }
            if (event.type === "session.error") {
              send("error", "Provider reported a session error.");
              break;
            }
          }
        } catch (error) {
          send("error", error instanceof Error ? error.message : "Run failed");
        }
        res.end();
        return;
      }
      const files: Record<string, string> = {
        "/": "index.html",
        "/style.css": "style.css",
        "/app.js": "app.js",
        "/astronaut.png": "astronaut.png",
      };
      if (req.method === "GET" && req.url && files[req.url]) {
        const f = files[req.url],
          data = await readFile(path.join(__dirname, "..", "public", f));
        res.writeHead(200, {
          "Content-Type": f.endsWith(".png")
            ? "image/png"
            : f.endsWith(".css")
              ? "text/css"
              : f.endsWith(".js")
                ? "text/javascript"
                : "text/html; charset=utf-8",
        });
        res.end(data);
        return;
      }
      res.writeHead(404).end("Not found");
    } catch (error) {
      if (!res.headersSent) res.writeHead(400, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: error instanceof Error ? error.message : "Request failed",
        }),
      );
    }
  })
  .listen(port, "127.0.0.1", () => console.log(`Open http://127.0.0.1:${port}`));

3 · Understand the lesson-specific rules · part 1

Runs a real session through a local event router. Action events are displayed, not executed or approved. Deduplication is scoped to one run.

Source: src/router.ts · starting at line 1

export type RoutedEvent = {
  type: string;
  id?: string;
  evaluated_permission?: string;
  [key: string]: unknown;
};
export type Sink = "preview" | "text" | "action" | "usage" | "lifecycle";
export function createRouter() {
  const seen = new Set<string>();
  return async (
    event: RoutedEvent,
    dispatch: (sink: Sink, event: RoutedEvent) => Promise<void> | void,
  ) => {
    if (event.type === "event_start" || event.type === "event_delta") {
      await dispatch("preview", event);
      return;
    }
    if (event.id && seen.has(event.id)) return;
    let sink: Sink = "lifecycle";
    if (event.type === "agent.message") sink = "text";
    else if (
      event.type === "agent.custom_tool_use" ||
      event.evaluated_permission === "ask"
    )
      sink = "action";
    else if (event.type === "span.model_request_end") sink = "usage";
    await dispatch(sink, event);
    if (event.id) seen.add(event.id);
  };
}

What the evidence proves

Runs a real session through a local event router. Action events are displayed, not executed or approved. Deduplication is scoped to one run.

Check the displayed outcome and source together. A rendered answer alone does not prove that a tool ran, a remote operation succeeded, or work was cancelled.
